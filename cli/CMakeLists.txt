# Carries OrcaSlicer code: the legacy-keys step below reads OrcaSlicer's
# src/libslic3r/PrintConfig.cpp (PrintConfigDef::handle_legacy) at configure time, and the runtime
# library list follows orcaslicer_copy_dlls in OrcaSlicer's root CMakeLists.txt (v2.4.2, commit
# 8500fcd). Added 2026-10-30. Those portions are copyright the OrcaSlicer contributors (NOTICE).
#
# sketch-orca: a command-line front end on OrcaSlicer's slicing library (libslic3r), with
# no GUI code. Orca's src/CMakeLists.txt adds this folder (patches/0003) when SLIC3R_GUI is
# OFF and SKETCH_ORCA_CLI_DIR points here; the binary folder is <build>/sketch-orca.
#
# The executable lands in <build>/sketch-orca next to every runtime library it loads (OpenCASCADE,
# FreeType, GMP, MPFR and the MSVC runtime on Windows), so that folder plus Orca's resources is
# the whole engine. Its commands and files are specified in protocol/README.md.

cmake_minimum_required(VERSION 3.13)

# The front end's own version; Orca's version and commit are reported next to it.
set(SKETCH_ORCA_VERSION "0.1.0")
if (NOT DEFINED SKETCH_ORCA_COMMIT)
    set(SKETCH_ORCA_COMMIT "")
endif ()
if (NOT DEFINED SKETCH_ORCA_TAG)
    set(SKETCH_ORCA_TAG "")
endif ()

set(SKETCH_ORCA_SOURCES
    main.cpp
    common.hpp
    common.cpp
    config_def.cpp
    gui_symbols.cpp
    job.hpp
    job.cpp
    presets.hpp
    presets.cpp
    run.hpp
    run.cpp
    stats.hpp
    stats.cpp
)
if (WIN32)
    list(APPEND SKETCH_ORCA_SOURCES sketch-orca.manifest)
endif ()

add_executable(sketch-orca ${SKETCH_ORCA_SOURCES})

# The keys Orca's PrintConfigDef::handle_legacy names, read from the Orca source the engine is built
# from: those it compares a setting's key with (opt_key == "...") and its set of ignored keys.
# config-def lists the ones it drops or renames whatever their value as obsolete (config_def.cpp),
# so the list is Orca's own at the locked commit and no copy of it is kept by hand. The file is
# written only when its keys change, and CMake runs again when PrintConfig.cpp changes.
set(_sketch_print_config "${TOP_LEVEL_PROJECT_DIR}/src/libslic3r/PrintConfig.cpp")
if (NOT EXISTS "${_sketch_print_config}")
    message(FATAL_ERROR "sketch-orca: ${_sketch_print_config} is missing; config-def reads Orca's legacy keys from it")
endif ()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_sketch_print_config}")
file(READ "${_sketch_print_config}" _sketch_text)
string(FIND "${_sketch_text}" "void PrintConfigDef::handle_legacy(" _sketch_from)
string(FIND "${_sketch_text}" "void PrintConfigDef::handle_legacy_composite(" _sketch_to)
if (_sketch_from EQUAL -1 OR _sketch_to LESS_EQUAL _sketch_from)
    message(FATAL_ERROR "sketch-orca: PrintConfigDef::handle_legacy, followed by handle_legacy_composite, is not in ${_sketch_print_config}")
endif ()
math(EXPR _sketch_length "${_sketch_to} - ${_sketch_from}")
string(SUBSTRING "${_sketch_text}" ${_sketch_from} ${_sketch_length} _sketch_legacy)
unset(_sketch_text)
string(REGEX MATCHALL "opt_key[ \t\r\n]*==[ \t\r\n]*\"[A-Za-z][A-Za-z0-9_]*\"" _sketch_compared "${_sketch_legacy}")
string(FIND "${_sketch_legacy}" "ignore = {" _sketch_ignore_at)
if (_sketch_ignore_at EQUAL -1)
    message(FATAL_ERROR "sketch-orca: handle_legacy in ${_sketch_print_config} has no set of ignored keys (ignore = { ... })")
endif ()
string(SUBSTRING "${_sketch_legacy}" ${_sketch_ignore_at} -1 _sketch_ignore)
string(FIND "${_sketch_ignore}" "};" _sketch_ignore_end)
if (_sketch_ignore_end EQUAL -1)
    message(FATAL_ERROR "sketch-orca: the set of ignored keys in handle_legacy (${_sketch_print_config}) does not end with };")
endif ()
string(SUBSTRING "${_sketch_ignore}" 0 ${_sketch_ignore_end} _sketch_ignore)
string(REGEX MATCHALL "\"[A-Za-z][A-Za-z0-9_]*\"" _sketch_ignored "${_sketch_ignore}")
list(LENGTH _sketch_compared _sketch_compared_count)
list(LENGTH _sketch_ignored _sketch_ignored_count)
if (_sketch_compared_count EQUAL 0 OR _sketch_ignored_count EQUAL 0)
    message(FATAL_ERROR "sketch-orca: no legacy keys found in handle_legacy (${_sketch_compared_count} compared, ${_sketch_ignored_count} ignored)")
endif ()
set(_sketch_keys "// Generated by sketch-orca's cli/CMakeLists.txt from Orca's src/libslic3r/PrintConfig.cpp\n// (PrintConfigDef::handle_legacy): ${_sketch_compared_count} keys it compares with, then ${_sketch_ignored_count} it ignores.\n")
foreach (_sketch_match IN LISTS _sketch_compared _sketch_ignored)
    string(REGEX MATCH "\"[A-Za-z][A-Za-z0-9_]*\"" _sketch_key "${_sketch_match}")
    string(APPEND _sketch_keys "${_sketch_key},\n")
endforeach ()
set(_sketch_inc_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(_sketch_inc "${_sketch_inc_dir}/legacy_keys.inc")
set(_sketch_old "")
if (EXISTS "${_sketch_inc}")
    file(READ "${_sketch_inc}" _sketch_old)
endif ()
if (NOT "${_sketch_old}" STREQUAL "${_sketch_keys}")
    file(WRITE "${_sketch_inc}" "${_sketch_keys}")
endif ()
message(STATUS "sketch-orca: ${_sketch_compared_count} keys handle_legacy compares with and ${_sketch_ignored_count} it ignores, in ${_sketch_inc}")
target_include_directories(sketch-orca PRIVATE "${_sketch_inc_dir}")

target_link_libraries(sketch-orca PRIVATE libslic3r cereal::cereal boost_libs boost_headeronly nlohmann_json)
target_compile_definitions(sketch-orca PRIVATE
    SKETCH_ORCA_VERSION="${SKETCH_ORCA_VERSION}"
    SKETCH_ORCA_COMMIT="${SKETCH_ORCA_COMMIT}"
    SKETCH_ORCA_TAG="${SKETCH_ORCA_TAG}"
)
set_target_properties(sketch-orca PROPERTIES
    OUTPUT_NAME "sketch-orca"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
)

if (WIN32)
    if (MSVC)
        # A console program; the manifest (UTF-8 code page, long paths) is merged by the linker.
        target_link_options(sketch-orca PRIVATE /SUBSYSTEM:CONSOLE)
    endif ()

    if ("${CMAKE_SYSTEM_PROCESSOR}" STREQUAL "ARM64")
        set(_sketch_arch "arm64")
    elseif ("${CMAKE_SYSTEM_PROCESSOR}" STREQUAL "X86")
        set(_sketch_arch "x86")
    else ()
        set(_sketch_arch "x64")
    endif ()

    # The same runtime libraries Orca copies next to its own executable (orcaslicer_copy_dlls in
    # Orca's root CMakeLists.txt), without WebView2, which only the GUI uses.
    set(_sketch_occt TKBO TKBRep TKCAF TKCDF TKernel TKG2d TKG3d TKGeomAlgo TKGeomBase TKHLR TKLCAF
        TKMath TKMesh TKPrim TKService TKShHealing TKSTEP TKSTEP209 TKSTEPAttr TKSTEPBase TKTopAlgo
        TKV3d TKVCAF TKXCAF TKXDESTEP TKXSBase)
    set(_sketch_dlls
        "${TOP_LEVEL_PROJECT_DIR}/deps/GMP/gmp/lib/win-${_sketch_arch}/libgmp-10.dll"
        "${TOP_LEVEL_PROJECT_DIR}/deps/MPFR/mpfr/lib/win-${_sketch_arch}/libmpfr-4.dll"
        "${CMAKE_PREFIX_PATH}/bin/freetype.dll"
    )
    foreach (_lib ${_sketch_occt})
        list(APPEND _sketch_dlls "${CMAKE_PREFIX_PATH}/bin/occt/${_lib}.dll")
    endforeach ()

    # The MSVC runtime, app-local, so the engine runs where the Visual C++ redistributable is not
    # installed.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
    set(CMAKE_INSTALL_UCRT_LIBRARIES FALSE)
    include(InstallRequiredSystemLibraries)
    list(APPEND _sketch_dlls ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS})

    foreach (_dll ${_sketch_dlls})
        if (NOT EXISTS "${_dll}")
            message(FATAL_ERROR "sketch-orca: runtime library ${_dll} is missing")
        endif ()
    endforeach ()
    add_custom_command(TARGET sketch-orca POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_sketch_dlls} "$<TARGET_FILE_DIR:sketch-orca>"
        COMMENT "Copying the runtime libraries next to sketch-orca"
        VERBATIM
    )
endif ()
